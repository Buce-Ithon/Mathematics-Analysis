\section{Basics of order, $\mathcal{O}$ and $o$}
\subsection{Summary/Comprehensions}
The knowledge of orders of $2$ infinitesimal or infinite quantities and the connection between them are common for us. 
Here we list some notes you may need to pay attention to: 

1. Sometimes the "big $\mathcal{O}$ constant" in a strong function can be related to the argument, and the same is true for the small 
"o". \\
\fbox{\begin{minipage}{0.95\textwidth}
    \begin{ex*}
        $\sin(xy) = O(1)$, $\sin(xy) = O_{y}(x)$
    \end{ex*}
\end{minipage}}

2. The infinitesimal of 2 functions can't always keep for their inverse functions, i.e. $\varphi(x)=o(f(x)) \nRightarrow 
\widetilde{f}(x)=o(\widetilde{\varphi}(x))$. (Here: $\varphi(x)$ and $f(x)$ are increasing functions, as $\widetilde{\varphi}(x)$ 
and $\widetilde{f}(x)$ are their inverse functions) \\
\fbox{\begin{minipage}{0.95\textwidth}
    \begin{ex*}
        $f(x)=e^{ x },\ \ \varphi(x)=\frac{e^{ x }}{x},\ \ x\to \infty$
    \end{ex*}
\end{minipage}}

3. The elementary of $\mathcal{O}$ and $o$ notation: \\
\fbox{\begin{minipage}{0.95\textwidth}
    \begin{nt*}
        $\varphi=O(f)$ means an inequality, while $\varphi=o(g)$ express a limitation.
    \end{nt*}
\end{minipage}}

4. 16 Rules of calculating $\mathcal{O}$ and $o$: \\
\fbox{\begin{minipage}{0.95\textwidth}
    \begin{enumerate}
        \item[1.] Transitivity
        \subitem[1] $\lim_{x\to x_0} f(x)=\infty$, $\varphi(x)=O(1)$ $\implies$ $\varphi(x)=o(f(x))$
        \subitem[2] $f=O(\varphi)$, $\varphi=O(\psi)$ $\implies$ $f=O(\psi)$ 
        \subitem[3] $f=O(\varphi)$, $\varphi=o(\psi)$ $\implies$ $f=o(\psi)$
        \item[2.] $\mathcal{O}$ 
        \subitem[4] $O(f)+O(g)=O(f+g)$ 
        \subitem[5] $O(f)O(g)=O(fg)$
        \item[3.] $\mathcal{O}$ \& $o$ 
        \subitem[6] $o(1)O(f)=o(f)$ 
        \subitem[7] $O(1)o(f)=o(f)$ 
        \subitem[8] $O(f)+o(f)=O(f)$ 
        \item[4.] $o$ 
        \subitem[9] $o(f)+o(g)=o(|f|+|g|)$ 
        \subitem[10] $o(f)o(g)=o(fg)$
        \item[5.] Power
        \subitem[11] $(O(f))^{k}=O_{k}(f^{k}),\ k\in\mathbb{N}$
        \subitem[12] $(o(f))^{k}=o(f^{k})$
        \item[6.] Equivalence $\sim$ 
        \subitem[13] $f\sim g$, $g\sim \varphi$ $\implies$ $f\sim \varphi$  
    \end{enumerate}
\end{minipage}}
\newline\newline
\fbox{\begin{minipage}{0.95\textwidth}
    \begin{enumerate}
        \item[6.] Equivalence $\sim$ 
        \subitem[14] $f=o(g)$, $g\sim \varphi$ $\implies$ $g\sim \varphi \pm kf$ 
        \item[7.] Summation of $o$ 
        \subitem[15] $f,\ g>0$, $f=o(g),\ x\to\infty$ $\implies$ $\int_{A}^{B}f(x)dx=o(\int_{A}^{B}g(x)dx)$,

5. Extension for Taylor's formula: \textcolor{skycyan}{/Tech/} 
\begin{thm}
    If $f^{(n)}(x)$ exists on $U(x_0;\delta)$ and $|f^{(n)}(x)| \leq M$, then $$f(x) = \sum_{k=0}^{n-1}\frac{f^{(k)}(x_0)}{k!}(x-x_0)^k+O(|(x-x_0)^n|)$$ holds on $U(x_0;\delta)$.
\end{thm}
\fbox{\begin{minipage}{0.95\textwidth}
    \begin{cor}
        \begin{enumerate}
            \item  $\sin x = x - \frac{x^3}{3!} + O(x^5)$, $x\to 0$ or $|x|<\infty$
            \item  $\cos x = 1 - \frac{x^2}{2!} + O(x^4)$, $x\to 0$ or $|x|<\infty$
            \item  $\log (1+x) = x - \frac{x^2}{2} + O(x^3)$, $x\to 0$
            \item  $(1+x)^{\alpha} = 1 + \alpha x + O(x^2)$, $x\to 0$
            \item  $e^x = 1 + x + O(x^2)$, $x\to 0$ or $|x|<1$
        \end{enumerate}
    \end{cor}

6. Abel type theorem: \textcolor{skycyan}{/Tech/} 
\begin{thm}
    \marginpar{\footnotesize $<$Abel type Theorem - 1$>$: \\(dicrete) $<$Stolz's Theorem$>$}
    Given $b_n>0$, $\sum_{n=1}^{\infty} b_n = \infty$, $a_n = o(b_n), n\to\infty$, then $$\sum_{n=1}^{N} a_n = o(\sum_{n=1}^{N} b_n),\ N\to\infty.$$
\end{thm}
\fbox{\begin{minipage}{0.95\textwidth}
    \begin{sol*}
        Denote $K_n$: the greatest $k$ s.t. $\sum_{i=1}^{k} b_i < \sqrt{\sum_{i=1}^{n} b_i}$ holds, then $K_n\to\infty, n\to\infty$. \\
        Then $\left\vert \sum_{i=1}^{n} a_i \right\vert = O(1) \sum_{i=1}^{K_n} b_i + o(1) \sum_{i=K_n}^{n} b_i = O(1) \sqrt{\sum_{i=1}^{n} b_i} + o(1) \sum_{i=1}^{n} b_i = o(1) \sum_{i=1}^{n} b_i$.
    \end{sol*}
    \begin{nt*}
        1. Comparing with ordinary $\epsilon-N$ method, the oder estimation proof method is concise and elegent. \\
        2. The commutive of $\sum$ and $o$.
    \end{nt*}
\end{minipage}}

7. $\Gamma$-function and Stirling's Formula. 
\begin{df}
    Given $x\in\mathbb{C}$, $\Gamma(x) = \int_{0}^{\infty} t^{x-1}e^{-t}dt,\ Re(x)>0$.
\end{df}
5 properties of $\Gamma(x)$: \\
\fbox{\begin{minipage}{0.95\textwidth}
    \begin{enumerate}
        \item $\Gamma(x) = (x-1)\Gamma(x-1) = \cdots = (x-1)\cdots(x-n)\Gamma(x-n)$ \\
        Specially, $\Gamma(n+1) = n!$.
        \item \textcolor{skycyan}{/Tech tool/} \\
        Lemma1. $\forall \alpha, \beta>0$, $\frac{\Gamma(\alpha)\Gamma(\beta)}{\Gamma(\alpha+\beta)} = \int_0^\infty \frac{t^{\beta-1}}{(1+t)^{\alpha+\beta}} dt = \int_0^1 y^{\alpha-1}(1-y)^{\beta-1} dy$.
        \fbox{\begin{minipage}{0.90\textwidth}
            Pre-Lemma 1. If $\forall b > a$, we have $$\int_a^b dx \int_\alpha^\beta f(x,y) dy = \int_\alpha^\beta dy \int_a^b f(x,y) dx,$$ where $a, \beta<\infty$, all are proper integrals; and if $\int_a^\infty f(x,y) dx$ converges uniformly for $\alpha \leq y \leq \beta$, then $$\int_a^\infty dx \int_\alpha^\beta f(x,y) dy = \int_\alpha^\beta dy \int_a^\infty f(x,y) dx.$$ \\
            Pre-Lemma 2. Suppose $f(x,y) \geq 0$. If $\forall c < b$ and $\gamma < \beta$, we have $$\int_a^c dx \int_\alpha^\beta f(x,y) dy = \int_\alpha^\beta dy \int_a^c f(x,y) dx$$ and $$\int_a^b dx \int_\alpha^\gamma f(x,y) dy = \int_\alpha^\gamma dy \int_a^b f(x,y) dx,$$ then when one of the following expressions exists, we have $$\int_a^b dx \int_\alpha^\beta f(x,y) dy = \int_\alpha^\beta dy \int_a^b f(x,y) dx;$$ when $b$ or $\beta$ is infinite, the above conclusion still holds.
        \end{minipage}}
        \item \textcolor{skycyan}{/Tech tool/} (Legendre duplication formula) \\
        Lemma 2. $\Gamma(2x)\Gamma\left(\frac{1}{2}\right) = 2^{2x-1}\Gamma(x)\Gamma\left(x + \frac{1}{2}\right)$, where $\Gamma(\frac{1}{2}) = \sqrt{\pi}$.
        \item \textcolor{skycyan}{/Tech tool/} \\
        Lemma 3. $\exists C$, s.t. $\log \Gamma(n) = (n-\frac{1}{2})\log n - n + C + O(\frac{1}{n})$. 
        \item \textcolor{skycyan}{/Tech tool/} \\
        Lemma 4. $\frac{\Gamma(x)}{\Gamma(x+a)} = x^{-a}+O(x^{-a-1})$.
    \end{enumerate}
\end{minipage}}
\newline\newline
The proof of some important of them will be displayed at the end of this chapter, hope you could try to feel and understand them, or even prove them by yourself.
\begin{thm}
    \textcolor{skycyan}{/Tech/} $$\Gamma(x) = x^{x-\frac{1}{2}}e^{-x}\sqrt{2\pi}(1+O(\frac{1}{x})),\ x>0.$$
\end{thm}
\fbox{\begin{minipage}{0.95\textwidth}
    \begin{sol*}
        Just condier $x\in\mathbb{R}/\mathbb{Z}$, since $\Gamma(x) \in C[0,\infty)$. Let $x = n + a$, where $n\in\mathbb{Z}, 0 < a < 1$, then
        \begin{equation*}
            \begin{split}
            \log \Gamma(x) &= \log \Gamma(n + a) = \log \Gamma(n) + a \log n + O(\frac{1}{n}) \ \text{(Lemma 4)}\\
            &= (n - \frac{1}{2}) \log n - n + C + a \log n + O(\frac{1}{n}) \ \text{(Lemma 3)}\\
            &= (x - a - \frac{1}{2}) \log (x - a) - x + a + C + a \log (x - a) + O(\frac{1}{x}) \\
            &= [x'\triangleq x-a] = (x - \frac{1}{2}) \log x - x + C + O(\frac{1}{x}) \ \text{(1)}
            \end{split}
        \end{equation*}
        Also $\log \Gamma(2x) + \log \Gamma(\frac{1}{2}) = (2x - 1) \log 2 + \log \Gamma(x) + \log \Gamma(x + \frac{1}{2})$ (Lemma 2), take the $\log \Gamma(x)$ expression (1) into LHS: 
        \begin{equation*}
            \begin{split}
            (2x - \frac{1}{2}) &\log (2x) - 2x + C + \log \Gamma(\frac{1}{2}) + O(\frac{1}{x}) \\
            &= (2x - 1) \log 2 + (x - \frac{1}{2}) \log x + x \log (x + \frac{1}{2}) \\
            &-2x - \frac{1}{2} + 2C + O(\frac{1}{x})
            \end{split}
        \end{equation*}
        Equating the constant terms on both sides, we get $C = \log \sqrt{2\pi}$, substituting into (1), we obtain the conclusion of the theorem.
    \end{sol*}
\end{minipage}}
\begin{cor}
    (Stirling's Formula) \textcolor{skycyan}{/Tech/} $$n! = n^{n + \frac{1}{2}} e^{-n} \sqrt{2\pi} (1 + O(\frac{1}{n})) = \sqrt{2\pi n} (\frac{n}{e})^n (1 + O(\frac{1}{n})).$$
\end{cor}
\fbox{\begin{minipage}{0.95\textwidth}
    \begin{sol*}
        $n! = \Gamma(n+1) = (n+1)^{n + \frac{1}{2}} e^{-(n+1)} \sqrt{2\pi} \left(1 + O\left(\frac{1}{n}\right)\right)$ (Thm 1.5) and
        \begin{align*}
            (n+1)^{n+\frac{1}{2}} &= n^{n+\frac{1}{2}} (1 + \frac{1}{n})^{n+\frac{1}{2}} = n^{n+\frac{1}{2}} (1 + \frac{1}{n})^n (1 + O(\frac{1}{n})) \\
            &= n^{n+\frac{1}{2}} e^{n \log(1+\frac{1}{n})} (1 + O(\frac{1}{n})) \\
            &= n^{n+\frac{1}{2}} e^{n (\frac{1}{n} + O(\frac{1}{n^2}))} (1 + O(\frac{1}{n})) \\
            &= n^{n+\frac{1}{2}} e^{1 + O(\frac{1}{n})} (1 + O(\frac{1}{n})) = n^{n+\frac{1}{2}} e (1 + O(\frac{1}{n})) 
    \end{align*}
    \end{sol*}
\end{minipage}}
\begin{nt*}
    1. Observing Stirling's Formula and previous estimation experience, we just need to estimate part $(n+1)^{n+\frac{1}{2}} = n^{n+\frac{1}{2}} (1+\frac{1}{n})^{n+\frac{1}{2}}$. (This kind of trick will be encountered many times.) \\
    2. Tips: $1^{\circ}.$ $(1+O(\frac{1}{n})+O(\frac{1}{n^2})) = 1+O(\frac{1}{n})$; $2^{\circ}.$ $(1+O(\frac{1}{n}))^k = 1+O(\frac{1}{n}),\ k\in\mathbb{Z}^{+}$; $3^{\circ}.$ $O(\frac{1}{n}) = o(1), n\to\infty$.
\end{nt*}
\fbox{\begin{minipage}{0.95\textwidth}
    Application of Corollary 1.6 (Stirling's Formula): \\
    Estimate the binomial coeffecient: $C_n^k = \frac{n!}{k!(n - k)!} \quad (k = 0, 1, \cdots, n)$.
    Substitute with Stirling's formula: $C_n^k = (\frac{n}{2\pi k(n-k)})^{\frac{1}{2}}(\frac{n}{k})^k(\frac{n}{n-k})^{n-k}(1+O(\frac{1}{k}))$. \\
    Let $k=\frac{n}{2}-l$, study the condition $l=o(n)$: \\
    since $(\frac{n}{k(n-k)})^{\frac{1}{2}} = (\frac{n}{(\frac{n}{2})^2 - l^2})^{\frac{1}{2}} = 2 (\frac{1}{n})^{\frac{1}{2}} (1 + O(\frac{l^2}{n^2}))$ and $(\frac{n}{k})^k (\frac{n}{n-k})^{n-k} = 2^n (1 - \frac{2l}{n})^{l - \frac{n}{2}} (1 + \frac{2l}{n})^{-l - \frac{n}{2}}$, then $C_n^k \sim ( \frac{2}{n\pi} )^{\frac{1}{2}} 2^n ( 1 + \frac{2l}{n} )^{-l - \frac{n}{2}} ( 1 - \frac{2l}{n} )^{l - \frac{n}{2}}$; 
    also $(1 - \frac{2l}{n})^{l - \frac{n}{2}} = e^(l - \frac{l^2}{n} + O (\frac{l^3}{n^2}))$ and $(1 + \frac{2l}{n})^{-l - \frac{n}{2}} = e^{(-l - \frac{l^2}{n} + O (\frac{l^3}{n^2}))}$, then $C_n^k \sim ( \frac{2}{n\pi} )^{\frac{1}{2}} 2^n e^{(-\frac{2l^2}{n} + O(\frac{l^3}{n^2}))}$. 
    In short, $C_n^k \begin{cases} \sim (\frac{2}{n\pi})^{\frac{1}{2}} 2^n ,\ l=0 \\ \sim C_n^{\frac{n}{2}} e^(-\frac{2l^2}{n} + O(\frac{l^3}{n^2})) ,\ l=o(n)\end{cases}$.
\end{minipage}}

    \marginpar{\footnotesize $\frac{a+o(1)}{b+o(1)} = \frac{a}{b}+o(1)$}
    Given $f(x) = o(1)$, $g(x) = o(1)$, $b\neq 0$, then 
    \begin{equation*}
        \begin{aligned}
                \frac{a + f(x)}{b + g(x)} &= \frac{a + f(x)}{b} \left(1 + \frac{g(x)}{b}\right)^{-1} = \frac{a + f(x)}{b} \left(1 - \frac{g(x)}{b} + O(g^2(x))\right) \\
                &= \frac{a}{b} + \frac{f(x)}{b} - \frac{ag(x)}{b^2} + O(f(x)g(x)) + O(g^2(x)).
        \end{aligned}
    \end{equation*}
\end{prb*}
