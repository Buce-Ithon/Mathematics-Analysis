\subsubsection{Conditions for curve integrals to be independent of the path}
\fbox{\begin{minipage}{0.95\textwidth}
\textcolor{skycyan}{Theorem}\\
In simply connected space $D$ and given a path $C:A\to B$, II-line integral $\int_{C} P \, dx+Q \, dy$ is dependent of the integration path $C:A\to B$ 
i.f.f. $\frac{\partial P}{\partial y}=\frac{\partial Q}{\partial x}$.
\end{minipage}}
\newline\newline
\fbox{\begin{minipage}{0.95\textwidth}
\textcolor{skycyan}{Proof idea}\\
Necessity "$\Rightarrow$": \\
By studying the differentiation method (partial differentiation with respect to $x$ and $y$) of $F(x, y) = \int_{(x_0, y_0)}^{(x, y)} P  dx + Q  dy$ and the primitive function of $P  dx + Q  dy$ (regarding it as the total differential of some function) (i.e., $\exists \Phi$ such that $P = \frac{\partial \Phi}{\partial x}$, $Q = \frac{\partial \Phi}{\partial y}$ and $\int_C P  dx + Q  dy = \Phi(x, y) \big|_{(x_A, y_A)}^{(x_B, y_B)}$), 
we can conclude that the necessary and sufficient condition for $\int_C P  dx + Q  dy$ to be path-independent is that $P  dx + Q  dy$ ($= d\Phi(x)$) is an exact differential. The latter implies $\frac{\partial P}{\partial y} = \frac{\partial Q}{\partial x}$ (necessity). \\
Sufficiency "$\Leftarrow$": \\
Fix $y \in [c, d]$, integrate $P = \frac{\partial \Phi}{\partial x}$ on $[x_0, x] \subset D$, then $\Phi(x, y) = \int_{x_0}^x P  dx + \Phi(x_0, y)$.
Also, fix $x = x_0$, integrate $Q = \frac{\partial \Phi}{\partial y}$ on $[y_0, y] \subset D$, then $\Phi(x_0, y) = \int_{y_0}^y Q(x_0, y)  dy + \Phi(x_0, y_0)$.\\
Then $\Phi(x, y) = \int_{x_0}^x P(x, y)  dx + \int_{y_0}^y Q(x_0, y)  dy + \Phi(x_0, y_0)$ or $\Phi(x, y) = \int_{y_0}^y Q(x, y)  dy + \int_{x_0}^x P(x, y_0)  dx + \Phi(x_0, y_0)$, which satisfies $\frac{\partial P}{\partial y} = \frac{\partial Q}{\partial x}$.
\end{minipage}}

\subsubsection{Calculate: $2$-dim Multiple Integral $\to$ Iterated Integral}
\fbox{\begin{minipage}{0.95\textwidth}
\textcolor{skycyan}{$2$-dim Multiple Integral $\to$ Iterated Integral}\\
\underline{Rectangle $P=[a,b]\times[c,d]$}: \\
If $\iint_P f(x,y) dP$ exists and $I_1=\int_a^b f(x,y) dx$ exists $\forall y \in [c,d]$, then
$\iint_P f(x,y) dP=\int_c^d dy \int_a^b f(x,y) dx\ .$\\
If $\iint_P f(x,y) dP$ exists and $I_2=\int_c^d f(x,y) dy$ exists $\forall x \in [a,b]$, then
$\iint_P f(x,y) dP=\int_a^b dx \int_c^d f(x,y) dy\ .$ \\
$\implies$ If $I_1$ and $I_2$ exist, then $\iint_P f(x,y) dP=\int_a^b dx \int_c^d f(x,y) dy=\int_c^d dy \int_a^b f(x,y) dx$.\\
\underline{Curved trapezoid $P=[a,b]\times[y_0(x),Y(x)]$ or $P=[x_0(y),X(y)]\times[c,d]$}: \\
If $\iint_P f(x,y) dP$ exists and $I_2=\int_{y_0(x)}^{Y(x)} f(x,y) dy$ exists $\forall x \in [a,b]$, then
$\iint_P f(x,y) dP=\int_a^b dx \int_{y_0(x)}^{Y(x)} f(x,y) dy\ .$\\
If $\iint_P f(x,y) dP$ exists and $I_1=\int_{x_0(y)}^{X(y)} f(x,y) dx$ exists $\forall y \in [c,d]$, then
$\iint_P f(x,y) dP=\int_c^d dy \int_{x_0(y)}^{X(y)} f(x,y) dx\ .$ \\
$\implies$ If $I_1$ and $I_2$ exist, then $\iint_P f(x,y) dP=\int_a^b dx \int_{y_0(x)}^{Y(x)} f(x,y) dy=\int_{y_0(x)}^{Y(x)} dy \int_a^b f(x,y) dx$.

\subsubsection{Green's Formula: $2$-dim Multiple Integral $\longleftrightarrow$ II-Line Integral}
\fbox{\begin{minipage}{0.95\textwidth}
\textcolor{skycyan}{Green's Formula}\\
$$\iint_{D} (\frac{\partial P}{\partial x}-\frac{\partial Q}{\partial y}) \, dx dy=\oint_{C=\partial D} P \, dy+Q \, dx$$
or 
$$\iint_{D} (\frac{\partial P}{\partial x}+\frac{\partial Q}{\partial y}) \, dx dy=\oint_{C=\partial D} P \, dy-Q \, dx$$
where $D$ is a simply connected domain in $\mathbb{R}^2$ with boundary $C=\partial D$ and $P, Q$ are continuously differentiable functions on $D$.
\end{minipage}}
\newline\newline
\fbox{\begin{minipage}{0.95\textwidth}
\textcolor{skycyan}{Proof idea}\\
$P, Q, \frac{\partial P}{\partial x}, \frac{\partial Q}{\partial y}$ continuously on $D$, $\partial D \triangleq C$.\\
$\iint_{D} \frac{\partial P}{\partial x} dx dy = \int_c^d dy \int_{x_0(y)}^{X(y)} \frac{\partial P}{\partial x} dx = \int_c^d [P(X(y), y) - P(x_0(y), y)] dy = \oint_{\partial D} P(x,y) dy$\\
$\iint_{D} \frac{\partial Q}{\partial y} dx dy = \int_a^b dx \int_{y_0(x)}^{Y(x)} \frac{\partial Q}{\partial y} dy = \int_a^b [Q(x, Y(x)) - Q(x, y_0(x))] dx = - \oint_{\partial D} Q(x,y) dx$

\end{figure}
\end{minipage}}
\subsubsection{Variable Substitution in Double Integrals}
\fbox{\begin{minipage}{0.95\textwidth}
\textcolor{skycyan}{Intro}\\
$D=\oint_{S=\partial D} x dy=\left[\begin{cases} x=x(\xi,\eta)\\ y=y(\xi,\eta) \end{cases}\right]=\oint_{\Sigma=\partial\Delta} x(\xi,\eta)\frac{\partial y}{\partial \xi} d\xi + x(\xi,\eta)\frac{\partial y}{\partial \eta} d\eta=\iint_{\Delta} \frac{D(x,y)}{D(\xi,\eta)} d\xi d\eta$
\end{minipage}}
\newline\newline
\fbox{\begin{minipage}{0.95\textwidth}
\textcolor{skycyan}{Analysis}\\
As we could see, the essential of $\begin{cases} x=x(\xi,\eta)\\ y=y(\xi,\eta) \end{cases}$ actually is coordinate transform between the $x,y$-plane and the $\xi,\eta$-plane.\\
$D=\oint_S x dy=\left[\begin{cases} x=x(\xi,\eta)\\ y=y(\xi,\eta) \end{cases},\begin{cases} \xi=\xi(t)\\ \eta=\eta(t) \end{cases} \right]=\int_{\alpha}^{\beta}x(\xi(t),\eta(t))(\frac{\partial y}{\partial\xi}\xi'(t)+\frac{\partial y}{\partial\eta}\eta'(t))dt=\oint_{\Sigma}x(\xi,\eta)\frac{\partial y}{\partial\xi}d\xi+x(\xi,\eta)\frac{\partial y}{\partial\eta}d\eta$,
let $x(\xi,\eta)\frac{\partial y}{\partial\xi} \triangleq P(x,y),\ x(\xi,\eta)\frac{\partial y}{\partial\eta} \triangleq Q(x,y)$, then $D=\oint_{\Sigma} P d\xi + Q d\eta=(\frac{\partial P}{\partial\eta}=\frac{\partial Q}{\partial\xi},\ \text{using Greeni's formula})= \iint_{\Delta}(\frac{\partial Q}{\partial \xi}-\frac{\partial P}{\partial \eta}) d\xi d\eta=\iint_{\Delta}\frac{D(x,y)}{D(\xi,\eta)} d\xi d\eta$.
\end{minipage}}

\textcolor{skycyan}{Idea of Analysis}: \\
Using $\begin{cases} x=x(\xi,\eta)=x(\xi(t),\eta(t))\\ y=y(\xi,\eta)=y(\xi(t),\eta(t)) \end{cases}$ (coordinate transform with parameterization), transform the II-line integration $\oint_Sx dy$ (or $-\oint_S y dx$, or $\frac{1}{2}\oint_S x dy - y dx$) on $x,y$-plane

calculate the double integral (often transform it to iterated integral $\int_{D\xi} d\xi \int_{\eta_0(\xi)}^{\eta(\xi)} (\frac{\partial Q}{\partial \xi}-\frac{\partial P}{\partial \eta})d\xi$, or others).\\

\subsection{The Riemann Integral over an $n$-Dimensional Interval}
\subsubsection{Comprehensions}
First, the definition of Riemann integral over $n$-dimention interval is simimiar (almost same) with $1$-dimension condition ($\int_{I}f(x)dx := \lim\limits_{\lambda(P)\to 0}\sigma(f,P,\xi)$),
$$\int_{I}f(x^1,\cdots,x^n)dx^1\cdots dx^n\ \text{or}\ \int\cdots\int_{I}f(x^1,\cdots,x^2)dx^1\cdots dx^n$$
Then, introduce the necessity condition of it ($f\in\mathcal{R}(I)\implies f\ \text{is bounded on}\ I$).
After that, book gives some properties (Lemmas) about the set of measure zero (same with $1$-dimension condition), and developes the Cantor Theorem (about uniform continuity of continuous functions on compact set).
At last, we're given 2 criterions (necessity and sufficiency) of the Riemann integral (Darboux Criterion, Lebesgue Criterion). 
\newline\newline
\fbox{\begin{minipage}{0.95\textwidth}
\begin{ex*}
Let $f:I\to\mathbb{R}$ is a continuous real-value function on $I\subset\mathbb{R}^{n-1}$, then it's graph is a set measure zero in $n$-dimension.
\end{ex*}
\end{minipage}}

/Development of Cantor Theorem/ If the relation $\omega(f;x) < w_0$ holds at each point of a compact set 
$K$ for the function $f : K \to \mathbb{R}$, then for every $\varepsilon > 0$ there exists $\delta > 0$ such

\end{lem}

\subsubsection{Problems Sets}
\begin{prb}
\begin{enumerate}
  \item[a)] Show that a set of measure zero has no interior points.
  \item[b)] Show that not having interior points by no means guarantees that a set is of measure zero.
  \item[c)] Construct a set having measure zero whose closure is the entire space $\mathbb{R}^n$.
  \item[d)] A set $E \subset I$ is said to have content zero if for every $\varepsilon > 0$ it can be covered by a finite system of intervals $I_1, \ldots, I_k$ such that $\sum_{i=1}^k |I_i| < \varepsilon$. Is every bounded set of measure zero a set of content zero?
  \item[e)] Show that if a set $E \subset \mathbb{R}^n$ is the direct product $\mathbb{R} \times e$ of the line $\mathbb{R}$ and a set $e \subset \mathbb{R}^{n-1}$ of $(n-1)$-dimensional measure zero, then $E$ is a set of $n$-dimensional measure zero.
\end{enumerate}
\end{prb}
\fbox{\begin{minipage}{0.95\textwidth}
\begin{pf*}
\begin{enumerate}
  \item[a)] [Proof by contradiction] If $E$ has a point $x_0$ as an interior point, then there exists a ball $B(x_0, r)$ such that $B(x_0, r) \subset E$. Since the ball has positive measure, it follows that $E$ cannot have measure zero.
  \item[b)] [Counterexample] Consider the irrational numbers $\mathbb{R} \setminus \mathbb{Q}$, which have no interior points in $\mathbb{R}$ but are not of measure zero.
  \item[c)] /inspired by b)/ Consider the set $E = \mathbb{Q}^n$, the set of all points in $\mathbb{R}^n$ with rational coordinates. The closure of $E$ is $\mathbb{R}^n$, and $E$ has measure zero since it can be covered by countably many intervals whose total length can be made arbitrarily small.
  \item[d)] [Principle] No, since the Lebesgue measure zero set can be covered by at most countable intervals (which means measure $\mu=o(1)\cdot\mathcal{O}(1)$), while a set of content zero can be covered by a finite number of intervals (which means the content (also a kind of measure) sometimes is $\mathcal{O}(1)\cdot\mathcal{O}(1)$). [Example] The $\mathbb{Q}\cap[0,1]$ and Cantor set are sets of measure zero but none of content zero.
  \item[e)] From the construction of the product measure $\lambda_n$ (Lebesgue measure of $\mathbb{R}^n$) /Lemma1/ if $A \in B(\mathbb{R}^k)$ and $B \in B(\mathbb{R}^{n-k})$, ($0 < k < n$), then $\lambda_n(A \times B) = \lambda_k(A) \lambda_{n-k}(B)$. So here (using also that /Lemma2/ if $E_m \uparrow E$, $\lim_{m \to +\infty} \lambda_n(E_m) = \lambda_n(E)$) $\lambda_n(E) = \lambda_n(\mathbb{R} \times e)$ $= \lim_{m \to +\infty} \lambda_n((-m, m) \times e)$ 
  $= \lim_{m \to +\infty} \lambda_1((-m, m)) \times \lambda_{n-1}(e)$ $= \lim_{m \to +\infty} \lambda_1((-m, m)) \times 0$ $= 0$.
\end{enumerate}
\begin{nt*}
1. The proof of e) /Lemma2/ could found in \textit{Real and Complex Analysis} by Walter Rudin, Theorem 1.19(Page 16), based on the (positive) measure definition (countable additive of disjoint countable measurable sets).
\end{nt*}
\end{pf*}
\end{minipage}}

\begin{prb}
\begin{enumerate}
  \item [a)] Construct the analogue of the Dirichlet function in $\mathbb{R}^{n}$ and show that a bounded function $f:I\to\mathbb{R}$ equal to zero at almost every point of the interval $I$ may still fail to belong to $\mathcal{R}(I)$.
  \item [b)] Show that if $f\in\mathcal{R}(I)$ and $f(x)=0$, at almost all points of the interval $I$, then $\int\limits_{I}f(x)\,{\rm d}x=0$.
\end{enumerate}
\end{prb}
\fbox{\begin{minipage}{0.95\textwidth}

  \item[b)] /by definition/ $\int_{I}f(x)dx=\lim\limits_{\lambda(P)\to 0}\sum\limits_{\xi_i\in P}f(\xi_i)\Delta(x_i)=\lim\limits_{\lambda(P)\to 0}\sum\limits_{\xi_i\in A\cap P}f(\xi_i)\Delta(x_i)+\lim\limits_{\lambda(P)\to 0}\sum\limits_{\xi_j\in B\cap P}f(\xi_j)\Delta(x_i)=\mathcal{O}(1)\cdot o(1)+o(1)\cdot\mathcal{O}(1)$, where $A=\{\cup[x_k,x_{k+1}]\subset P:\forall k,\ \exists \xi\in[x_k,x_{k+1}],\ s.t.\ f(\xi)\neq 0\}$ and $B=I\backslash A$. 
\end{enumerate}
\end{pf*}

\begin{prb}
There is a small difference between our earlier definition of the Riemann integral on a closed interval $I \subset \mathbb{R}$ and Definition 7 for the integral over an interval of arbitrary dimension. This difference involves the definition of a partition and the measure of an interval of the partition. Clarify this nuance for yourself and verify that
$$\int_a^b f(x) dx = \begin{cases} \int_I f(x) dx & \text{ if } a < b \\ - \int_I f(x) dx & \text{ if } a > b \end{cases}$$
where $I$ is the interval on the real line $\mathbb{R}$ with endpoints $a$ and $b$.
\end{prb}
\fbox{\begin{minipage}{0.95\textwidth}
\begin{pf*}

\begin{prb}
\begin{enumerate}
  \item [a)] Prove that a real-valued function $f : I \to \mathbb{R}$ defined on an interval $I \subset \mathbb{R}^n$ is integrable over that interval if and only if for every $\varepsilon > 0$ there exists a partition $P$ of $I$ such that $S(f; P) - s(f; P) < \varepsilon$.
  \item [b)] Using the result of a) and assuming that we are dealing with a real-valued function $f : I \to \mathbb{R}$, one can simplify slightly the proof of the sufficiency of the Lebesgue criterion. Try to carry out this simplification by yourself.
\end{enumerate}
\end{prb}
